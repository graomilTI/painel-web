import { supabase } from "./supabaseClient.js"; import { loadUserContext, getUserContext, logout } from "./authContext.js"; // =============================== // INIT // =============================== async function init() { const { data } = await supabase.auth.getSession(); if (!data?.session) { window.location.href = "/painel/login.html"; return; } try { await loadUserContext(); renderUser(); renderMenu(); bindEvents(); } catch (err) { console.error("Erro ao iniciar dashboard:", err); window.location.href = "/painel/login.html"; } } // =============================== // USER INFO // =============================== function renderUser() { const ctx = getUserContext(); if (!ctx) return; const nome = ctx.usuario.nome; const perfil = ctx.usuario.perfil_nome; document.getElementById("welcomeUser").innerText = `Olá, ${nome}`; document.getElementById("roleBadge").innerText = perfil.toUpperCase(); } // =============================== // MENU DINÂMICO // =============================== function renderMenu() { const ctx = getUserContext(); if (!ctx) return; const container = document.getElementById("sidebarMenu"); if (!container) return; const modulos = ctx.modulos || []; // ordena modulos.sort((a, b) => a.ordem - b.ordem); container.innerHTML = modulos .map( (m) => ` ${m.nome} ` ) .join(""); } // =============================== // EVENTOS // =============================== function bindEvents() { document.getElementById("signOutBtn")?.addEventListener("click", logout); } // =============================== init();
